```python
def _apply_drummer_profile(self, piano_roll: np.ndarray, profile: str) -> np.ndarray:
    """Apply drummer-specific characteristics"""
    
    if profile.lower() == 'jeff_porcaro':
        # Jazz ride, ghost notes
        piano_roll[4] *= 1.2  # Ride boost
        piano_roll[1] = np.where(piano_roll[1] < 0.3, 
                                 piano_roll[1] * 0.5, piano_roll[1])
    
    elif profile.lower() == 'steve_gadd':
        # Linear, dynamics
        piano_roll *= 1.1
        piano_roll = np.clip(piano_roll, 0, 1)
    
    elif profile.lower() == 'bernard_purdie':
        # Shuffle feel
        for i in range(1, 128, 2):
            if i < 127:
                piano_roll[:, i+1] *= 0.8
    
    # NEW PROFILES
    elif profile.lower() == 'john_bonham':
        # Power, heavy bass drum
        piano_roll[0] *= 1.3  # Kick louder
        piano_roll[5] *= 1.2  # Toms prominent
        # Add triplets (every 3rd 16th note)
        for i in range(0, 128, 3):
            if piano_roll[0, i] > 0.5:
                piano_roll[0, i] = min(1.0, piano_roll[0, i] * 1.2)
    
    elif profile.lower() == 'stewart_copeland':
        # Hi-hat mastery, linear
        piano_roll[2] *= 1.3  # Hi-hat boost
        piano_roll[3] *= 1.2  # Open hi-hat
        piano_roll[0] *= 0.7  # Less bass drum
        # Ensure linear (reduce overlaps)
        for i in range(128):
            if np.sum(piano_roll[:, i] > 0.3) > 2:
                # Reduce to 2 simultaneous hits max
                active = np.where(piano_roll[:, i] > 0.3)[0]
                if len(active) > 2:
                    for lane in active[2:]:
                        piano_roll[lane, i] *= 0.5
    
    elif profile.lower() == 'phil_collins':
        # Gated reverb, melodic fills
        # Boost toms for fills
        piano_roll[5] *= 1.3
        # Add gated reverb flag (for export)
        # Clean timing (reduce micro-variations in humanization)
        pass
    
    elif profile.lower() == 'clyde_stubblefield':
        # Funky Drummer characteristics
        piano_roll[2] *= 1.2  # Hi-hat prominent
        # Ghost notes on snare
        for i in range(128):
            if piano_roll[1, i] < 0.4 and piano_roll[1, i] > 0.1:
                piano_roll[1, i] *= 0.6  # Soften ghosts
        piano_roll[0] *= 0.8  # Less bass drum
    
    elif profile.lower() == 'vinnie_colaiuta':
        # Technical precision
        # High dynamics, clean execution
        piano_roll *= 1.15
        piano_roll = np.clip(piano_roll, 0, 1)
    
    return piano_roll
```
